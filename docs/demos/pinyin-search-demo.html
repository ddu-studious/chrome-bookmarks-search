推荐方案：方案A (pinyin-match)

  • 体积最优：仅 27KB，gzip 后 19KB，不影响扩展加载速度
  • 功能匹配：支持首字母、全拼、混合拼音搜索，正好满足书签搜索需求
  • 位置信息：返回匹配位置 [start, end]，可直接用于搜索结果高亮
  • 零依赖：纯 JS 实现，一个文件引入即可
  • 多音字：内置多音字支持，如 "长" (cháng/zhǎng)
  • 集成简单：只需在 SearchParser.matchAllKeywords 中增加拼音匹配分支

集成方式

// search-parser.js 中的改动示意

static matchAllKeywords(item, exactMatches, keywords, inField, excludeExact, excludeKeywords) {
  let searchable = ...; // 现有逻辑不变

  // 现有：纯文本匹配
  for (const kw of keywords) {
    if (!searchable.includes(kw.toLowerCase())) {
      // 新增：如果文本不匹配，尝试拼音匹配
      const title = item.title || '';
      const pinyinResult = PinyinMatch.match(title, kw);
      if (!pinyinResult) {
        return false; // 文本和拼音都不匹配
      }
    }
  }
  return true;
}
